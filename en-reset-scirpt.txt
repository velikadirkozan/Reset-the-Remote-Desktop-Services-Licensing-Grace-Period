
# ============================================================
# RDS Grace Period Checker
# ============================================================
# Description : Checks the remaining Remote Desktop Services
#               (RDS) licensing grace period in Windows Server.
#
# Developer   : Veli Kadir Kozan
# Email       : kadir@kadirkozan.com
# GitHub      : https://github.com/velikadirkozan
#
# Supported OS:
# Windows Server 2012 R2 / 2016 / 2019 / 2022
#
# Note        : This script only checks the grace period.
#               It does not reset or extend RDS licensing.
# ============================================================

#Requires -RunAsAdministrator

try {
    Write-Host "`n========================================" -ForegroundColor Cyan
    Write-Host "       RDS GRACE PERIOD CHECKER" -ForegroundColor Cyan
    Write-Host "========================================" -ForegroundColor Cyan

    Write-Host "`nChecking RDS licensing grace period..." -ForegroundColor Yellow

    # Retrieve Remote Desktop Services configuration
    $RDS = Get-CimInstance `
        -Namespace "root/CIMV2/TerminalServices" `
        -ClassName "Win32_TerminalServiceSetting" `
        -ErrorAction Stop

    # Retrieve remaining grace period days
    $Result = Invoke-CimMethod `
        -InputObject $RDS `
        -MethodName "GetGracePeriodDays" `
        -ErrorAction Stop

    # Validate the method execution
    if ($Result.ReturnValue -ne 0) {
        throw "Failed to retrieve RDS grace period. Error code: $($Result.ReturnValue)"
    }

    $Days = [int]$Result.DaysLeft

    # Display server information
    Write-Host "`n----------------------------------------" -ForegroundColor DarkGray
    Write-Host "Server Name    : $env:COMPUTERNAME" -ForegroundColor White
    Write-Host "Remaining Days : $Days" -ForegroundColor White
    Write-Host "----------------------------------------" -ForegroundColor DarkGray

    # Evaluate remaining grace period
    if ($Days -le 0) {
        Write-Host "`nSTATUS: EXPIRED" -ForegroundColor Red
        Write-Host "The RDS licensing grace period has expired or requires verification." -ForegroundColor Red
        Write-Host "Configure a valid RDS License Server and install the required CALs." -ForegroundColor Yellow
    }
    elseif ($Days -le 15) {
        Write-Host "`nSTATUS: CRITICAL" -ForegroundColor Red
        Write-Host "The RDS grace period is about to expire!" -ForegroundColor Red
        Write-Host "Complete the required licensing configuration immediately." -ForegroundColor Yellow
    }
    elseif ($Days -le 30) {
        Write-Host "`nSTATUS: WARNING" -ForegroundColor Yellow
        Write-Host "The RDS grace period will expire soon." -ForegroundColor Yellow
        Write-Host "Ensure that the RDS licensing configuration is completed." -ForegroundColor White
    }
    else {
        Write-Host "`nSTATUS: ACTIVE" -ForegroundColor Green
        Write-Host "The RDS licensing grace period is still active." -ForegroundColor Green
    }

    Write-Host "`n========================================" -ForegroundColor Cyan
    Write-Host "RDS Grace Period Check Completed" -ForegroundColor Cyan
    Write-Host "========================================`n" -ForegroundColor Cyan
}
catch {
    Write-Host "`nERROR: $($_.Exception.Message)" -ForegroundColor Red
    Write-Host "Unable to retrieve RDS licensing information." -ForegroundColor Yellow
    exit 1
}
